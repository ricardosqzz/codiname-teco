  135  # Sha
  136  def sha(p):
  137      h=hashlib.sha256()
  138      with p.open('rb') as f:
  139          for b in iter(lambda:f.read(4*1024*1024),b''):h.update(b)
  140      return h.hexdigest()

  142  # Polygon
  143  def polygon(g):
  144      if g is None or g.is_empty:return g
  145      g=make_valid(g)
  146      if g.geom_type in ['Polygon','MultiPolygon']:return g
  147      return unary_union([v for v in g.geoms if v.geom_type in ['Polygon','MultiPolygon']])

  184      assert all(len(c)<=10 for c in cols),cols
