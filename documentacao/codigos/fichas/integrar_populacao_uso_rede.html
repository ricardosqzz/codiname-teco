  162      def __init__(self,extent):
  163          self.ds=rasterio.open(LULC);assert self.ds.crs.to_epsg()==CRS and self.ds.count==1
  164          w=from_bounds(*extent,transform=self.ds.transform);c=math.floor(w.col_off)-1;r=math.floor(w.row_off)-1
  165          w=Window(c,r,math.ceil(w.col_off+w.width)-c+1,math.ceil(w.row_off+w.height)-r+1)
  166          self.window=w;self.transform=self.ds.window_transform(w);self.a=self.ds.read(1,window=w,boundless=True,fill_value=0)
  167          self.cell_area=abs(self.transform.a*self.transform.e);self.codes=np.unique(self.a);self.codes=self.codes[self.codes>0]
  168          print('Raster de análise',self.a.shape,'classes',self.codes.tolist(),flush=True)
  169      def add(self,g,level):
  170          n=len(g);lab=rasterize([(p,i+1) for i,p in enumerate(g.geometry)],out_shape=self.a.shape,transform=self.transform,fill=0,all_touched=False,dtype='int32')
  171          total=np.bincount(lab.ravel(),minlength=n+1)[1:]
  172          hist={int(c):np.bincount(lab[self.a==c],minlength=n+1)[1:] for c in self.codes}
  173          valid=sum(hist.values());g=g.copy();g['uso_ano']=2025;g['uso_npix']=total;g['uso_valid']=valid
  174          g['uso_cov']=np.divide(100*valid,total,out=np.full(n,np.nan),where=total>0);g['uso_km2']=valid*self.cell_area/1e6
  175          assigned=np.zeros(n,dtype=np.int64);stack=[];labels=[]
  176          for prefix,(codes,label) in GROUPS.items():
  177              count=sum((hist.get(c,np.zeros(n,dtype=np.int64)) for c in codes));assigned+=count
  178              g[f'{prefix}_km2']=count*self.cell_area/1e6;g[f'pct_{prefix}']=np.divide(100*count,valid,out=np.full(n,np.nan),where=valid>0)
  179              stack.append(count);labels.append(label)
  180          other=valid-assigned;assert (other>=0).all()
  181          g['out_km2']=other*self.cell_area/1e6;g['pct_out']=np.divide(100*other,valid,out=np.full(n,np.nan),where=valid>0)
  182          stack.append(other);labels.append('Outros códigos');top=np.argmax(np.stack(stack),axis=0)
  183          g['uso_dom']=[labels[i] if v>0 else 'Sem dado de uso' for i,v in zip(top,valid)]
  184          rows=[]
  185          for i,(_,row) in enumerate(g.iterrows()):
  186              for code,counts in hist.items():
  187                  if counts[i]>0:rows.append(dict(escala=level,setor_id=row.get('setor_id',''),bas_id=row.get('bas_id',''),codigo=code,classe=CLASS.get(code,f'Classe {code}'),n_pixels=int(counts[i]),area_km2=float(counts[i]*self.cell_area/1e6),pct_valid=float(100*counts[i]/valid[i])))
  188          assert np.allclose(g[[f'pct_{p}' for p in GROUPS]+['pct_out']].sum(axis=1)[valid>0],100)
  189          del lab;gc.collect();return g,rows
  190      def write_clip(self,bas):
  191          mask=rasterize([(g,1) for g in bas.geometry],out_shape=self.a.shape,transform=self.transform,fill=0,dtype='uint8')
  192          out=np.where(mask,self.a,0)
  193          rows,cols=np.where(mask>0);r0,r1=rows.min(),rows.max()+1;c0,c1=cols.min(),cols.max()+1
  194          profile=self.ds.profile.copy();profile.update(height=r1-r0,width=c1-c0,transform=self.transform*rasterio.Affine.translation(c0,r0),nodata=0,count=1,compress='LZW',driver='GTiff')
  195          with rasterio.open(OUT/'uso_ocupacao_bacias.tif','w',**profile) as d:
  196              d.write(out[r0:r1,c0:c1],1);d.update_tags(ANO_INFORMADO_PROJETO='2025',ANO_CONFIRMADO_METADADOS='NAO',RECORTE='Bacias TECO; regra centro de pixel; sem reamostragem',SOURCE=str(LULC))

  200      if g is None or g.is_empty:return []
  201      if g.geom_type in ['LineString','LinearRing']:return [g] if g.length>1e-7 else []
  202      if hasattr(g,'geoms'):return [p for a in g.geoms for p in lines(a)]

  206  def network_layers(bas,bh_domain):
  207      source=gpd.read_file(NETWORK).to_crs(CRS);source['sit_rede']=source.IND_RDESG.fillna('NI').replace({'':'NI'})
  208      assert set(source.sit_rede.unique())<=set(['S','N','NI'])
  209      target=bas.geometry.union_all().intersection(bh_domain)
  210      take=source.sindex.query(target,predicate='intersects');r=source.iloc[take].copy();r.geometry=r.geometry.intersection(target);r=r[r.length>1e-7].copy()
  211      unions={k:unary_union(r.loc[r.sit_rede==k].geometry.tolist()) for k in ['S','N','NI']}
  212      conflict=unions['S'].intersection(unions['N']);clean={'S':unions['S'].difference(unions['N']),'N':unions['N'].difference(unions['S']),'NI':unions['NI'].difference(unions['S'].union(unions['N'])),'C':conflict}
  213      parts=[];names={'S':'Com rede cadastrada','N':'Sem rede cadastrada','NI':'Sem informação','C':'Conflito S/N na origem'}
  214      for key,g in clean.items():
  215          for p in lines(g):parts.append(dict(sit_rede=key,sit_nome=names[key],geometry=p))
  216      net=gpd.GeoDataFrame(parts,crs=CRS);net=gpd.overlay(net,bas[['bas_id','bas_nome','geometry']],how='intersection',keep_geom_type=True)
  217      net=net.explode(index_parts=False,ignore_index=True);net=net[net.length>1e-7].copy();net['comp_m']=net.length
  218      unique=unary_union(r.geometry.tolist()).length
  219      assert abs(net['comp_m'].sum()-unique)<max(.01,unique*1e-8),(net.comp_m.sum(),unique)
  220      raw=gpd.overlay(r[['ID_BASE_TR','ID_RDESG','LADO_RDESG','DATA','sit_rede','geometry']],bas[['bas_id','bas_nome','geometry']],how='intersection',keep_geom_type=True)
  221      raw=raw.rename(columns={'ID_BASE_TR':'id_trecho','ID_RDESG':'id_cad','LADO_RDESG':'lado','DATA':'data_apur'})
  222      raw['data_apur']=raw['data_apur'].map(lambda t:None if pd.isna(t) else pd.Timestamp(t).strftime('%Y-%m-%d'));raw['comp_m']=raw.length
  223      audit=dict(source_features=len(source),source_flags=source.sit_rede.value_counts().to_dict(),source_duplicates_wkb=int(source.geometry.to_wkb().duplicated().sum()),raw_features_in_work_area=len(raw),consolidated_parts=len(net),unique_length_km=unique/1000,conflict_length_km=clean['C'].length/1000,unknown_overlap_known_removed_km=(unions['NI'].length-clean['NI'].length)/1000,rule='Unique geometrical linework; S/N coincident portions flagged C, not assigned automatically; known status overrides NI only on exact overlap',dictionary_url='https://ckan.pbh.gov.br/dataset/cb4df0e5-0849-4d1f-9f19-32a189824a7a/resource/329ad4ec-5912-457d-b13a-134747a2e55a/download/trecho_rede_esgoto.pdf')
  224      print('Cadastro PBH consolidado',json.dumps(audit,ensure_ascii=True),flush=True)
  225      return net,raw,audit

  227  # Add network
  228  def add_network(g,net,bh_domain):
  229      g=g.copy();statuscols={'S':'rede_s_km','N':'rede_n_km','NI':'rede_i_km','C':'rede_c_km'}
  230      fields=list(statuscols.values())+['rede_t_km','pct_t_s','cad_comp','dens_rede']
  231      for c in fields:g[c]=np.nan
  232      g['area_bh']=g.geometry.intersection(bh_domain).area/1e6;g['bh_pct']=100*g['area_bh']/(g.area/1e6);g['rede_qual']='fora_BH'
  233      for idx,row in g.iterrows():
  234          if row['area_bh']<=1e-10:continue
  235          geom=row.geometry.intersection(bh_domain);ids=net.sindex.query(geom,predicate='intersects')
  236          values={k:0. for k in statuscols}
  237          if len(ids):
  238              candidate=net.iloc[ids];lengths=length(intersection(candidate.geometry.to_numpy(),geom))/1000
  239              for key in values:values[key]=float(np.asarray(lengths)[candidate.sit_rede.to_numpy()==key].sum())
  240          for key,col in statuscols.items():g.at[idx,col]=values[key]
  241          total=sum(values.values());g.at[idx,'rede_t_km']=total;g.at[idx,'dens_rede']=values['S']/row['area_bh'];g.at[idx,'rede_qual']='cadastro' if total>0 else 'sem_tr'
  242          if total>0:g.at[idx,'pct_t_s']=100*values['S']/total;g.at[idx,'cad_comp']=100*(values['S']+values['N'])/total
  243      return g
