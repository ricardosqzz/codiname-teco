  236  def field_alias(name):
  237   if name in FIELDS:return FIELDS[name]
  238   if name.startswith('I_S'):return 'Relative pressure score - '+name[2:]
  239   if name.startswith('F_S'):return 'Relative pressure class - '+name[2:]

  241   if name.startswith('N_'):return 'Normalised '+field_alias(name[2:]).lower()
  242   return name
